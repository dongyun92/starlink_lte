```python
def map_samples_to_altitude_phases(df):
    """샘플 인덱스를 정확한 고도 및 비행 단계로 매핑"""
    phase_map = {
        'ground_takeoff': df[df['altitude'] < 30].index.tolist(),
        'climb': df[(df['altitude'] >= 30) & (df['altitude'] < 90)].index.tolist(),
        'cruise_high': df[df['altitude'] >= 90].index.tolist(),
    }

    altitude_stats = {
        phase: {
            'samples': len(indices),
            'altitude_range': (df.loc[indices, 'altitude'].min(),
                             df.loc[indices, 'altitude'].max()),
            'lte_rssi_mean': df.loc[indices, 'lte_rssi'].mean(),
            'starlink_latency_mean': df.loc[indices[df.loc[indices, 'starlink_available']],
                                          'starlink_latency'].mean()
        }
        for phase, indices in phase_map.items() if len(indices) > 0
    }

    return phase_map, altitude_stats
```
